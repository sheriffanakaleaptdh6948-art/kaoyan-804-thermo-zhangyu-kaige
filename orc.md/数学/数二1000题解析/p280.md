$$
\iint_{D}\ln(xy)\mathrm{d}x\mathrm{d}y=\int_{\frac{1}{2}}^{\frac{\sqrt{2}}{2}}\mathrm{d}x\int_{\frac{1}{x}}^{4x}\ln(xy)\mathrm{d}y+\int_{\frac{\sqrt{2}}{2}}^{1}\mathrm{d}x\int_{\frac{1}{x}}^{\frac{2}{x}}\ln(xy)\mathrm{d}y+\int_{1}^{\sqrt{2}}\mathrm{d}x\int_{x}^{\frac{2}{x}}\ln(xy)\mathrm{d}y

=I_{1}+I_{2}+I_{3},
$$
其中 $\int\ln(xy)\mathrm{d}y=y\ln(xy)-y+C$，则
$$
\int_{\frac{1}{x}}^{4x}\ln(xy)\mathrm{d}y=4(2\ln 2-1)x+8x\ln x+\frac{1}{x},
$$
$$
\int_{\frac{1}{x}}^{\frac{2}{x}}\ln(xy)\mathrm{d}y=\frac{2\ln 2-1}{x},
$$
$$
\int_{x}^{\frac{2}{x}}\ln(xy)\mathrm{d}y=\frac{2\ln 2-2}{x}-2x\ln x+x,
