(2) 求混合气体的压力 $p^{\prime}$

$\begin{aligned}p^{\prime} & =\frac{n R T^{\prime}}{V}=\frac{\left(n_{\mathrm{O}_{2}}+n_{\mathrm{CO}_{2}}\right) R T^{\prime}}{n_{\mathrm{O}_{2}} R T_{\mathrm{O}_{2}} / p_{\mathrm{O}_{2}}+n_{\mathrm{CO}_{2}} R T_{\mathrm{CO}_{2}} / p_{\mathrm{CO}_{2}}} \\& =\frac{(2+3) \times 10^{3} \mathrm{~mol} \times 367.7 \mathrm{~K}}{2 \times 10^{3} \mathrm{~mol} \times 300 \mathrm{~K} /\left(5 \times 10^{5} \mathrm{~Pa}\right)+3 \times 10^{3} \mathrm{~mol} \times 400 \mathrm{~K} /\left(3 \times 10^{5} \mathrm{~Pa}\right)} \\& =3.54 \times 10^{5} \mathrm{~Pa}\end{aligned}$

(3) 热力学能变化为 $\Delta U=0$

(4) 焓的变化

$\begin{aligned}\Delta H & =\Delta H_{\mathrm{O}_{2}}+\Delta H_{\mathrm{CO}_{2}} \\& =n_{\mathrm{O}_{2}} C_{p, \mathrm{~m}, \mathrm{O}_{2}}\left(T^{\prime}-T_{\mathrm{O}_{2}}\right)+n_{\mathrm{CO}_{2}} C_{p, \mathrm{~m}, \mathrm{CO}_{2}}\left(T^{\prime}-T_{\mathrm{CO}_{2}}\right) \\& =2 \times 10^{3} \mathrm{~mol} \times \frac{7}{2} \times 8.314 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \times(367.7 \mathrm{~K}-300 \mathrm{~K})+ \\& 3 \times 10^{3} \mathrm{~mol} \times \frac{9}{2} \times 8.314 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \times(367.7 \mathrm{~K}-400 \mathrm{~K}) \\& =314.7 \times 10^{3} \mathrm{~J}=314.7 \mathrm{kJ}\end{aligned}$

(5) 熵的变化

混合后氧和二氧化碳的分压力分别为

$\begin{aligned}& p_{\mathrm{O}_{2}}^{\prime}=x_{\mathrm{O}_{2}} p^{\prime}=\frac{2}{5} \times 3.514 \times 10^{5} \mathrm{~Pa}=1.416 \times 10^{5} \mathrm{~Pa} \\& p_{\mathrm{CO}_{2}}^{\prime}=x_{\mathrm{CO}_{2}} p^{\prime}=\frac{3}{5} \times 3.54 \times 10^{5} \mathrm{~Pa}=2.124 \times 10^{5} \mathrm{~Pa}\end{aligned}$

$\begin{aligned}\Delta S & =n_{\mathrm{O}_{2}}\left(\Delta S_{\mathrm{m}}\right)_{\mathrm{O}_{2}}+n_{\mathrm{CO}_{2}}\left(\Delta S_{\mathrm{m}}\right)_{\mathrm{CO}_{2}} \\& =n_{\mathrm{O}_{2}}\left(C_{p, \mathrm{~m}, \mathrm{O}_{2}} \ln \frac{T^{\prime}}{T_{\mathrm{O}_{2}}}-\mathrm{R} \ln \frac{p_{\mathrm{O}_{2}}^{\prime}}{p_{\mathrm{O}_{2}}}\right)+n_{\mathrm{CO}_{2}}\left(C_{p, \mathrm{~m}, \mathrm{CO}_{2}} \ln \frac{T^{\prime}}{T_{\mathrm{CO}_{2}}}-\mathrm{R} \ln \frac{p_{\mathrm{CO}_{2}}^{\prime}}{p_{\mathrm{CO}_{2}}}\right) \\& =2 \times 10^{3} \mathrm{~mol} \times\left[\frac{7}{2} \times 8.314 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \ln \frac{367.7 \mathrm{~K}}{300 \mathrm{~K}}-\right. \\& \left.8.314 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \ln \frac{1.416 \times 10^{5} \mathrm{~Pa}}{5 \times 10^{5} \mathrm{~Pa}}\right]+\end{aligned}$
