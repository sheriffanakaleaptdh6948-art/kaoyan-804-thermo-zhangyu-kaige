$$
T_2 = T_1 \times \left(\frac{p_2}{p_1}\right)^{\frac{n-1}{n}}
$$
$$
= 305K \times \left(\frac{0.51Mpa}{0.102Mpa}\right)^{\frac{1.28-1}{1.28}}
$$
$$
= 433.71K
$$

$$
\therefore \Delta h = c_p(T_2 - T_1)
$$
$$
= 1.004kJ/(kg \cdot K) \times (433.71K - 305K)
$$
$$
= 129.22kJ/kg
$$

$$
\Delta s = c_p \ln\frac{T_2}{T_1} - R_g \ln\frac{p_2}{p_1}
$$
$$
= 1.004kJ/(kg \cdot K) \times \ln\frac{433.71K}{305K} - 0.287kJ/(kg \cdot K) \times \ln\frac{0.51Mpa}{0.102Mpa}
$$
$$
= -0.108kJ/(kg \cdot K)
$$

$$
\therefore w_t = \frac{n}{n-1} R_g T_1 \left[ 1 - \left(\frac{p_2}{p_1}\right)^{\frac{n-1}{n}} \right]
$$
$$
= \frac{1.28}{1.28-1} \times 0.287kJ/(kg \cdot K) \times 305K \times \left[ 1 - \left(\frac{0.51Mpa}{0.102Mpa}\right)^{\frac{1.28-1}{1.28}} \right]
$$
$$
= -168.87kJ/kg
$$

$$
\therefore q = \Delta h + w_t
$$
$$
= 129.22kJ/kg - 168.87kJ/kg
$$
$$
= -39.65kJ/kg
$$

$$
\therefore q_h = |q_m| |q| \times 3600
$$
$$
= 0.012kg/s \times 39.65kJ/kg \times 3600
$$
$$
= 1712.88kJ/h
$$
