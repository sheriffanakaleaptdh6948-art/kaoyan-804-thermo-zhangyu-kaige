$$
m\Delta T = \frac{\Delta H}{c_{\text{p}}} = - \frac{67.95}{5.20} = - 13.07 \text{ kg}\cdot\text{K}
$$
$$
c_{\text{v}} = \frac{\Delta U}{m\Delta T} = - \frac{40.81}{-13.07} = 3.12 \frac{\text{kJ}}{\text{kg}\cdot\text{K}}
$$
$$
R_{\text{g}} = c_{\text{p}} - c_{\text{v}} = 5.20 - 3.12 = 2.08 \frac{\text{kJ}}{\text{kg}\cdot\text{K}}
$$

$$
\Delta S_{\text{sys}} = \frac{Q}{T} = 100 \times \frac{335}{273} = 122.71 \frac{\text{kJ}}{\text{K}}
$$
$$
\text{熵流} S_{\text{f}} = \frac{Q}{T_0} = \frac{33500}{293} = 114.33 \frac{\text{kJ}}{\text{K}}
$$
$$
\text{熵产} S_{\text{g}} = \Delta S_{\text{sys}} - S_{\text{f}} = 122.71 - 114.33 = 8.38 \text{ kJ/K}
$$
$$
\text{㶲损失} I = T_0 S_{\text{g}} = 293 \times 8.38 \approx 2455 \text{ kJ}
$$
