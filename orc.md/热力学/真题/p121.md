$$ \therefore c_p - c_v = -T \cdot \left(\frac{R_g}{p}\right)^2 \cdot \left[ - \frac{R_g T}{(v-b)^2} \right] $$
$$ = \frac{T^2 R_g^3}{p^2(v-b)^2} = \frac{T^2 R_g^3}{T^2 R_g^2} = R_g \text{为常数} $$

(1)

$$ \because T_2 = T_1 = 305K $$
$$ \therefore \Delta h = c_p(T_2 - T_1) = 0 $$
$$ c_p = c_v + R_g = 0.717kJ/(kg \cdot K) + 0.287kJ/(kg \cdot K) = 1.004kJ/(kg \cdot K) $$
$$ \Delta s = -R_g \ln \frac{p_2}{p_1} = -0.287kJ/(kg \cdot K) \times \ln \frac{0.51Mpa}{0.102Mpa} = -0.462kJ/(kg \cdot K) $$
$$ w_t = - \int_1^2 v dp = - \int_1^2 \frac{R_g T}{p} dp = - R_g T \ln \frac{p_2}{p_1} $$
$$ = -0.287kJ/(kg \cdot K) \times 305K \times \ln \frac{0.51Mpa}{0.102Mpa} $$
$$ = -140.88kJ/kg $$
$$ q = \Delta h + w_t = w_t = -140.88kJ/kg $$
$$ \therefore q_h = q_m \cdot |q| \times 3600 = 0.012kg/s \times 140.88kJ/kg \times 3600 $$
$$ = 6086.016kJ/h $$

(2)

$$ T_2 = T_1 \times \left(\frac{p_2}{p_1}\right)^{\frac{k-1}{k}} $$
$$ = 305K \times \left(\frac{0.51}{0.102}\right)^{\frac{1.4-1}{1.4}} $$
$$ = 483.06K $$
$$ \therefore \Delta h = c_p(T_2 - T_1) $$
$$ = 1.004kJ/(kg \cdot K) \times (483.06K - 305K) $$
$$ = 178.77kJ/kg $$

$$ \Delta s = c_p \ln \frac{T_2}{T_1} - R_g \ln \frac{p_2}{p_1} $$
$$ = 1.004kJ/(kg \cdot K) \times \ln \frac{483.06K}{305K} - 0.287kJ/(kg \cdot K) \times \ln \frac{0.51Mpa}{0.102Mpa} $$
$$ = -2.4 \times 10^{-4}kJ/(kg \cdot K) $$

$$ \because q = \Delta h + w_t, q = 0 $$
$$ \therefore w_t = - \Delta h = -178.77kJ/kg $$
$$ q_h = 0 $$
