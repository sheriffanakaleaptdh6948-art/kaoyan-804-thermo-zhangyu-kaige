$s^{\prime \prime}=8.149 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K}), s^{\prime}=0.649 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K})$

$s_{\mathrm{b}}=s_{2}, s^{\prime}

$x_{2}=\frac{s_{2}-s_{3}^{\prime}}{s^{\prime \prime}-s_{3}^{\prime}}=\frac{7.571 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K})-0.649 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K})}{8.149 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K})-0.649 \mathrm{~kJ} /(\mathrm{kg} \cdot \mathrm{K})}=0.923$

$h_{2}=h^{\prime}+x_{2}\left(h^{\prime \prime}-h^{\prime}\right)$
