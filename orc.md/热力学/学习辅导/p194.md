【例 7-4】某缩放喷管进口截面积为 $A_{1}=2.6 \times 10^{-3} \mathrm{~m}^{2}$ 。质量流量为 $q_{m}=$ $1.5 \mathrm{~kg} / \mathrm{s}$ 的空气等熵流经喷管, 进口截面上的压力和温度分别为 $p_{1}=$ $0.58 \mathrm{MPa} 、 T_{1}=440 \mathrm{~K}$, 已知喷管出口截面上压力 $p_{2}=0.14 \mathrm{MPa}$, 求喉部及出口截面积和出口流速。空气作理想气体处理, 比热容取定值, $R_{\mathrm{g}}=287 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K})$, $c_{p}=1004 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K})$ 。

解 $v_{1}=\frac{R_{\mathrm{g}} T_{1}}{p_{1}}=\frac{287 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K}) \times 440 \mathrm{~K}}{0.58 \times 10^{6} \mathrm{~Pa}}=0.2177 \mathrm{~m}^{3} / \mathrm{kg}$

$c_{\mathrm{f} 1}=\frac{q_{m} v_{1}}{A_{1}}=\frac{1.5 \mathrm{~kg} / \mathrm{s} \times 0.2177 \mathrm{~m}^{3} / \mathrm{kg}}{2.6 \times 10^{-3} \mathrm{~m}^{2}}=125.6 \mathrm{~m} / \mathrm{s}$

$T_{0}=T_{1}+\frac{c_{\mathrm{f} 1}^{2}}{2 c_{p}}=440 \mathrm{~K}+\frac{(125.6 \mathrm{~m} / \mathrm{s})^{2}}{2 \times 1004 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K})}=447.86 \mathrm{~K}$

$p_{0}=p_{1}\left(\frac{T_{0}}{T_{1}}\right)^{\frac{\kappa}{\kappa-1}}=0.58 \mathrm{MPa} \times\left(\frac{447.86 \mathrm{~K}}{440 \mathrm{~K}}\right)^{\frac{1.4-1}{1.4-1}}=0.617 \mathrm{MPa}$

$p_{\mathrm{cr}}=\nu_{\mathrm{cr}} p_{0}=0.528 \times 0.617 \mathrm{MPa}=0.326 \mathrm{MPa}$

$T_{\mathrm{cr}}=T_{0}\left(\frac{P_{\mathrm{cr}}}{p_{0}}\right)^{\frac{\kappa-1}{\kappa}}=T_{0} \nu_{\mathrm{cr}}^{\frac{\kappa-1}{\kappa}}=447.86 \mathrm{~K} \times 0.528^{\frac{1.4-1}{1.4-1}}=373.16 \mathrm{~K}$

$c_{\mathrm{t}, \mathrm{cr}}=\sqrt{\kappa R_{\mathrm{g}} T_{\mathrm{cr}}}=\sqrt{1.4 \times 287 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K}) \times 373.16 \mathrm{~K}}=387.2 \mathrm{~m} / \mathrm{s}$

$v_{\mathrm{cr}}=\frac{R_{\mathrm{g}} T_{\mathrm{cr}}}{p_{\mathrm{cr}}}=\frac{287 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K}) \times 373.16 \mathrm{~K}}{0.326 \times 10^{6} \mathrm{~Pa}}=0.329 \mathrm{~m}^{3} / \mathrm{kg}$

$A_{\mathrm{cr}}=\frac{q_{m} v_{\mathrm{cr}}}{c_{\mathrm{t}, \mathrm{cr}}}=\frac{1.5 \mathrm{~kg} / \mathrm{s} \times 0.329 \mathrm{~m}^{3} / \mathrm{kg}}{387.2 \mathrm{~m} / \mathrm{s}}=1.275 \times 10^{-3} \mathrm{~m}^{2}$

$T_{2}=T_{0}\left(\frac{p_{2}}{p_{0}}\right)^{\frac{\kappa-1}{\kappa}}=T_{0} \nu_{\mathrm{cr}}^{\frac{\kappa-1}{\kappa}}=447.86 \mathrm{~K} \times\left(\frac{0.14 \mathrm{MPa}}{0.617 \mathrm{MPa}}\right)^{\frac{1.4-1}{1.4-1}}=293.16 \mathrm{~K}$

$v_{2}=\frac{R_{\mathrm{g}} T_{2}}{p_{2}}=\frac{287 \mathrm{~J} /(\mathrm{kg} \cdot \mathrm{K}) \times 293.16 \mathrm{~K}}{0.14 \times 10^{6} \mathrm{~Pa}}=0.601 \mathrm{~m}^{3} / \mathrm{kg}$
