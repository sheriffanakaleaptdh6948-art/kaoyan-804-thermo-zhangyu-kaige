据 $p_{3}=0.8 \mathrm{MPa}, h_{3}=2831.64 \mathrm{~kJ} / \mathrm{kg}$, 查 $h-s$ 图得

$$
t_{3}=196.7{ }^{\circ} \mathrm{C}, v_{3}=0.2586 \mathrm{~m}^{3} / \mathrm{kg}
$$

查有关资料, 氮 $p_{\mathrm{cr}}=3.39 \mathrm{MPa}, T_{\mathrm{cr}}=126 \mathrm{~K}$

$$
\begin{aligned}
p_{1} & =20 \mathrm{MPa}, p_{2}=2 \mathrm{MPa}, T_{1}=(-70+273) \mathrm{K}=203 \mathrm{~K} \\
p_{\mathrm{r} 1} & =\frac{p_{1}}{p_{\mathrm{cr}}}=\frac{20 \mathrm{MPa}}{3.39 \mathrm{MPa}}=5.9, T_{\mathrm{r} 1}=\frac{T_{1}}{T_{\mathrm{cr}}}=\frac{203 \mathrm{~K}}{126 \mathrm{~K}}=1.61 ; \\
p_{\mathrm{r} 2} & =\frac{p_{2}}{p_{\mathrm{cr}}}=\frac{2 \mathrm{MPa}}{3.39 \mathrm{MPa}}=0.59
\end{aligned}
$$

查通用焓图

$$
\begin{aligned}
\left(\frac{H_{\mathrm{m}}^{*}-H_{\mathrm{m}}}{R T_{\mathrm{cr}}}\right)_{1} & =1.79 \\
\left(H_{\mathrm{m}}^{*}-H_{\mathrm{m}}\right)_{1} & =1.79 R T_{\mathrm{cr}} \\
& =1.79 \times 8.3145 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \times 126 \mathrm{~K} \\
& =1875.3 \mathrm{~J} / \mathrm{mol}
\end{aligned}
$$

假设 $T_{2}=140 \mathrm{~K}$, 则

$$
\begin{aligned}
T_{\mathrm{r} 2} & =\frac{T_{2}}{T_{\mathrm{cr}}}=\frac{140 \mathrm{~K}}{126 \mathrm{~K}}=1.11 \\
\left(\frac{H_{\mathrm{m}}^{*}-H_{\mathrm{m}}}{R T_{\mathrm{cr}}}\right)_{2} & =0.57 \\
\left(H_{\mathrm{m}}^{*}-H_{\mathrm{m}}\right)_{2} & =0.57 R T_{\mathrm{cr}}=0.57 \times 8.3145 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \times 126 \mathrm{~K} \\
& =597.1 \mathrm{~J} / \mathrm{mol}
\end{aligned}
$$

据绝热节流特征, $H_{\mathrm{m} 1}=H_{\mathrm{m} 2}$

$$
\begin{aligned}
H_{\mathrm{m} 2}-H_{\mathrm{m} 1} & =\left(H_{\mathrm{m} 1}^{*}-H_{\mathrm{m} 1}\right)+\left(H_{\mathrm{m} 2}^{*}-H_{\mathrm{m} 1}^{*}\right)-\left(H_{\mathrm{m} 2}^{*}-H_{\mathrm{m} 2}\right) \\
& =\left(H_{\mathrm{m} 1}^{*}-H_{\mathrm{m} 1}\right)-\left(H_{\mathrm{m} 2}^{*}-H_{\mathrm{m} 2}\right)+C_{p \mathrm{~m}}\left(T_{2}-T_{1}\right) \\
& =1875.3 \mathrm{~J} / \mathrm{mol}-597.1 \mathrm{~J} / \mathrm{mol}+29.1 \mathrm{~J} /(\mathrm{mol} \cdot \mathrm{K}) \times(140-203) \mathrm{K} \\
& =-555.1 \mathrm{~J} / \mathrm{mol}
\end{aligned}
