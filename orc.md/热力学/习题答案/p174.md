$\begin{aligned}& x_{\mathrm{CO}_{2}}=\frac{R}{M_{\mathrm{CO}_{2}} R_{\mathrm{g}}} w_{\mathrm{CO}_{2}}=\frac{8.3145}{44.01 \times 10^{-3} \times 288} \times 0.056=0.037 \\& x_{\mathrm{O}_{2}}=\frac{R}{M_{\mathrm{H}_{2} \mathrm{O}} R_{\mathrm{g}}} w_{\mathrm{O}_{2}}=\frac{8.3145}{32 \times 10^{-3} \times 288} \times 0.163=0.147 \\& x_{\mathrm{H}_{2} \mathrm{O}}=\frac{R}{M_{\mathrm{O}_{2}} R_{\mathrm{g}}} w_{\mathrm{H}_{2} \mathrm{O}}=\frac{8.3145}{18.016 \times 10^{-3} \times 288} \times 0.02=0.032 \\& x_{\mathrm{N}_{2}}=\frac{R}{M_{\mathrm{N}_{2}} R_{\mathrm{g}}} w_{\mathrm{N}_{2}}=\frac{8.3145}{28.02 \times 10^{-3} \times 288} \times 0.761=0.784\end{aligned}$
