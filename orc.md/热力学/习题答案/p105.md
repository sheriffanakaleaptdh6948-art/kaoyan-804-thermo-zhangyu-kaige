$=0.0313 \times 601+(0.72864-0.1) \times 10^{6} \times 2.8 \times 10^{-6}=20.6 \mathrm{~N}$

$M a=\frac{c_{1}}{c}=\frac{\sqrt{c_{\mathrm{p}}\left(T_{0}-T\right)}}{\sqrt{k R_{\mathrm{g}} T}}$

$T=\frac{2 c_{\mathrm{p}} T_{0}}{k R_{\mathrm{g}} M_{2}^{2}+2 c_{\mathrm{p}}}=\frac{2 \times 1005 \times 350}{1.4 \times 287 \times 0.6^{2}+2 \times 1005}=326.5 \mathrm{~K}$

$p=p_{0}\left(\frac{T}{T_{0}}\right)^{\frac{k}{k-1}}=0.65 \times\left(\frac{326.5}{350}\right)^{\frac{1.4-1}{1.4-1}}=0.51 \mathrm{MPa}$

$v=\frac{R_{\mathrm{g}} T}{p}=\frac{287 \times 326.5}{0.51 \times 10^{6}}=0.1839 \mathrm{~m}^{3} / \mathrm{kg}$

$c_{1}=\sqrt{2 \times 1005 \times(350-326.5)}=217.34 \mathrm{~m} / \mathrm{s}$

$q_{\mathrm{m}}=\frac{A c_{1}}{v}=\frac{2.6 \times 10^{-3} \times 217.34}{0.1839}=3.07 \mathrm{~kg} / \mathrm{s}$

$p_{\mathrm{cr}}=\nu_{\mathrm{cr}} p_{0}=0.528 \times 0.65=0.3432 \mathrm{MPa}>p_{\mathrm{b}}$

所以 $p_{2}=p_{\mathrm{cr}}=0.3432 \mathrm{MPa}$, 出口截面即为临界截面

$T_{2}=T_{\mathrm{cr}}=T_{0} \nu_{\mathrm{cr}} \frac{t-1}{k}=350 \times 0.528 \frac{t-1}{1.4-1}=291.62 \mathrm{~K}$

$v_{2}=\frac{R_{\mathrm{g}} T_{2}}{p_{2}}=\frac{287 \times 291.62}{0.3432 \times 10^{6}}=0.2439 \mathrm{~m}^{3} / \mathrm{kg}$

$c_{12}=\sqrt{2\left(h_{0}-h_{2}\right)}=\sqrt{2 c_{\mathrm{p}}\left(T_{0}-T_{2}\right)}$

$=\sqrt{2 \times 1005 \times(350-291.62)}=342.55 \mathrm{~m} / \mathrm{s}$

$A_{2}=\frac{q_{\mathrm{m}} v_{2}}{c_{12}}=\frac{3.07 \times 0.2439}{342.55}=2.19 \times 10^{-3} \mathrm{~m}^{2}$

$v_{1}=\frac{R_{\mathrm{g}} T_{1}}{p_{1}}=\frac{287 \times 440}{0.58 \times 10^{6}}=0.2177 \mathrm{~m}^{3} / \mathrm{kg}$

$c_{\mathrm{f}}=\frac{q_{\mathrm{m}} v_{1}}{A_{1}}=\frac{1.5 \times 0.2177}{2.6 \times 10^{-3}}=125.61 \mathrm{~m} / \mathrm{s}$

$T_{0}=T_{1}+\frac{c_{\mathrm{h}}^{2}}{2 c_{\mathrm{p}}}=440+\frac{125.61}{2 \times 1005}=447.85 \mathrm{~K}$
