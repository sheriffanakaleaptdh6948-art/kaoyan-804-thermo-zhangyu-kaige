$\begin{aligned} W=W_{1}+W_{2} & =\int_{V_{1}}^{V_{\mathrm{m}}} p \mathrm{~d} V+\int_{V_{\mathrm{m}}}^{V_{2}} p \mathrm{~d} V \\ & =\int_{V_{1}}^{V_{\mathrm{m}}}\left(0.4-0.5 V\right) \mathrm{d} V+\int_{V_{\mathrm{m}}}^{V_{2}} p_{2} \mathrm{~d} V \\ & =0.4 \times\left(V_{\mathrm{m}}-V_{1}\right)-\frac{0.5}{2}\left(V_{\mathrm{m}}^{2}-V_{1}^{2}\right)+p_{2} \times\left(V_{2}-V_{\mathrm{m}}\right) \\ & =\left[0.4 \times(0.6-0.4)-0.25 \times\left(0.6^{2}-0.4^{2}\right)+0.1 \times(0.8-0.6)\right] \times 10^{6} \\ & =1.5 \times 10^{4} \mathrm{~J} \end{aligned}$
